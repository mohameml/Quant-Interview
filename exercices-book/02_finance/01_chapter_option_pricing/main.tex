\chapter{Option pricing}

\section{Prix du Call européen dans le modèle de Black-Scholes}

\begin{exerciseBox}[Prix du Call européen dans le modèle de Black-Scholes]
Soit un actif \( S_t \) évoluant selon la dynamique :
\[
dS_t = r S_t\, dt + \sigma S_t\, dW_t
\]

où \( r \) est le taux sans risque constant, \( \sigma > 0 \), et \( (W_t) \) est un mouvement brownien sous la mesure risque-neutre.

\vspace{1cm}

On considère une option Call européenne de maturité \( T \) et de strike \( K \). 

\vspace{1cm}

\textbf{Question :} Montrer que le prix \( C_0 \) du Call à \( t=0 \) est donné par :
\[
C_0 = S_0 \Phi(d_1) - K e^{-rT} \Phi(d_2)
\]

avec :
\[
d_1 = \frac{\ln(S_0/K) + (r + \frac{\sigma^2}{2})T}{\sigma \sqrt{T}}, 
\quad 
d_2 = d_1 - \sigma \sqrt{T}
\]
\end{exerciseBox}


\subsection*{Solution :}

Sous la mesure risque-neutre \( \mathbb{Q} \), l’actif sous-jacent suit :

\[
dS_t = r S_t\, dt + \sigma S_t\, dW_t^{\mathbb{Q}}, \quad S_0 > 0
\]

La solution de cette EDS est :

\[
S_T = S_0 \cdot \exp\left( \left(r - \frac{\sigma^2}{2}\right)T + \sigma W_T^{\mathbb{Q}} \right)
\]

Le prix du call européen à \( t = 0 \) est donné par :

\[
C_0 = e^{-rT} \cdot \mathbb{E}^{\mathbb{Q}}\left[ (S_T - K)_+ \right]
\]


\subsection*{Décomposition de l’espérance}

\[
(S_T - K)_+ = S_T \cdot \mathbf{1}_{S_T > K} - K \cdot \mathbf{1}_{S_T > K}
\]

\[
\Rightarrow C_0 = e^{-rT} \left( \mathbb{E}[S_T \cdot \mathbf{1}_{S_T > K}] - K \cdot \mathbb{E}[ \mathbf{1}_{S_T > K}] \right)
\]


\subsection*{1. Calcul de \( \mathbb{E}[ \mathbf{1}_{S_T > K}] \)}

On a :


\begin{align*}
\mathbb{P}(S_T > K) 
&= \mathbb{P}\left( \ln\left( \frac{S_T}{K} \right) > 0 \right) 
\quad \text{(\(\ln\) est croissante)} \\
&= \mathbb{P}\left( \ln(S_T) > \ln(K) \right) \\
&= \mathbb{P}\left( \ln(S_0) + \left(r - \frac{\sigma^2}{2}\right)T + \sigma W_T^{\mathbb{Q}} > \ln(K) \right) 
 \\
&= \mathbb{P}\left( \sigma W_T^{\mathbb{Q}} > \ln(K) - \ln(S_0) - \left(r - \frac{\sigma^2}{2}\right)T \right) \\
&= \mathbb{P}\left( W_T^{\mathbb{Q}} > \frac{1}{\sigma} \left[ \ln(K/S_0) - \left(r - \frac{\sigma^2}{2}\right)T \right] \right) \\
&= \mathbb{P}\left( W_T^{\mathbb{Q}} > -d_2 \cdot \sqrt{T} \right) 
\quad \text{(on pose \( d_2 := \frac{\ln(S_0/K) + (r - \frac{\sigma^2}{2})T}{\sigma \sqrt{T}} \))} \\
&= \mathbb{P}\left( \frac{W_T^{\mathbb{Q}}}{\sqrt{T}} > -d_2 \right) 
\quad \text{(car \( W_T^{\mathbb{Q}} \sim \mathcal{N}(0, T) \))} \\
&= \mathbb{P}(G > -d_2), \quad G \sim \mathcal{N}(0,1) \\
&= 1 - \Phi(-d_2) \quad \text{avec $\Phi$ est la F.D.R du loi $\mathcal{N}(0,1)$} \\
&= \Phi(d_2) , \quad \Phi(-x) + \Phi(x) = 1
\end{align*}

% \newpage

\subsection*{2. Calcul de \( \mathbb{E}[S_T \cdot \mathbf{1}_{S_T > K}] \)}

\begin{rappelBox}[Théorème de Girsanov]
Soit \( (W_t)_{t \geq 0} \) un mouvement brownien sous \( \mathbb{P} \), et \( (h_t)_{t \geq 0} \) un processus progressif, tel que :
\[
Z_t := \exp\left( -\int_0^t h_s\, dW_s - \frac{1}{2} \int_0^t h_s^2\, ds \right)
\]
soit une \( \mathbb{P} \)-martingale.

Alors, la mesure \( \mathbb{Q} \) définie par \( \left. \frac{d\mathbb{Q}}{d\mathbb{P}} \right|_{\mathcal{F}_t} = Z_t \) transforme :
\[
\widetilde{W}_t := W_t + \int_0^t h_s\, ds
\quad \text{en un Brownien sous } \mathbb{Q}.
\]
\end{rappelBox}

\textbf{Formellement}, posons :



\begin{align*}
\mathbb{E}[S_T \cdot \mathbf{1}_{S_T > K}] 
&= \mathbb{E}^{\mathbb{Q}}\left[S_0 \cdot \exp\left( \left(r - \frac{\sigma^2}{2}\right)T + \sigma W_T^{\mathbb{Q}} \right) \cdot \mathbf{1}_{S_T > K} \right] \\
&= S_0 \cdot e^{rT} \cdot \mathbb{E}^{\mathbb{Q}}\left[ \exp\left( \sigma W_T^{\mathbb{Q}} - \frac{\sigma^2}{2}T \right) \cdot \mathbf{1}_{S_T > K} \right] \\
&= S_0 \cdot e^{rT} \cdot \mathbb{E}^{\mathbb{Q}}\left[ Z_T \cdot \mathbf{1}_{S_T > K} \right]
\quad \text{avec } Z_T := \exp\left( \sigma W_T^{\mathbb{Q}} - \frac{1}{2}\sigma^2 T \right) \\
&= S_0 \cdot e^{rT} \cdot \mathbb{E}^{\mathbb{Q}}\left[ \frac{d\mathbb{P}}{d\mathbb{Q}} \cdot \mathbf{1}_{S_T > K} \right]
\quad \text{(remarque : Girsanov avec } h_t = \sigma) \\
&= S_0 \cdot e^{rT} \cdot \mathbb{P}(S_T > K) \quad \text{car} \; \mathbb{E}^{\mathbb{Q}}\left[\frac{d\mathbb{P}}{d\mathbb{Q}}Y\right] = \mathbb{E}^{\mathbb{P}}(Y) \\
&= S_0 \cdot e^{rT} \cdot \mathbb{P}\left( \ln(S_T/K) > 0 \right) \\
&= S_0 \cdot e^{rT} \cdot \mathbb{P}\left( \ln(S_T) > \ln(K) \right) \\
&= S_0 \cdot e^{rT} \cdot \mathbb{P}\left( \ln(S_0) + \left(r - \frac{\sigma^2}{2} \right)T + \sigma W_T^{\mathbb{Q}} > \ln(K) \right) \\
&= S_0 \cdot e^{rT} \cdot \mathbb{P}\left( W_T^{\mathbb{Q}} > \frac{ \ln(K/S_0) - \left(r - \frac{\sigma^2}{2} \right)T }{\sigma} \right) \\
&= S_0 \cdot e^{rT} \cdot \mathbb{P}\left( W_T^{\mathbb{Q}} - \sigma T > \frac{ \ln(K/S_0) - \left(r - \frac{\sigma^2}{2} \right)T }{\sigma} - \sigma T \right) \\
&= S_0 \cdot e^{rT} \cdot \mathbb{P}\left( W_T^{\mathbb{Q}} - \sigma T > \frac{ \ln(K/S_0) - \left(r + \frac{\sigma^2}{2} \right)T }{\sigma} \right)  \quad \text{or} \; W_T^{\mathbb{Q}} - \sigma T \sim \mathcal{N}(0,T) \; \text{sous} \; \mathbb{P} \\
&= S_0 \cdot e^{rT} \cdot \mathbb{P}\left( \frac{W_T^{\mathbb{Q}} - \sigma T}{\sqrt{T}} > -d_1 \right)
\quad \text{(on pose \( d_1 := \frac{\ln(S_0/K) + (r + \frac{\sigma^2}{2})T}{\sigma \sqrt{T}} \))} \\
&= S_0 \cdot e^{rT} \cdot \mathbb{P}(G > -d_1), \quad \text{où } G \sim \mathcal{N}(0, 1) \\
&= S_0 \cdot e^{rT} \cdot \Phi(d_1)
\end{align*}


\subsection*{Conclusion}

On obtient :

\[
C_0 = e^{-rT} \left( S_0. e^{rT} \cdot \Phi(d_1) - K \cdot \Phi(d_2) \right)
= \boxed{ S_0 \Phi(d_1) - K e^{-rT} \Phi(d_2) }
\]

avec :
\[
d_1 = \frac{\ln(S_0/K) + (r + \frac{\sigma^2}{2})T}{\sigma \sqrt{T}}, \quad
d_2 = d_1 - \sigma \sqrt{T}
\]



\section{Option Chooser}

\subsection*{Exercice :}

\begin{exerciseBox}[Option Chooser]
On considère une option \emph{chooser} définie comme suit : 
à la date $T_1$, le détenteur choisit si l’option devient un call européen ou un put européen.
L’option choisie porte sur le même actif sous-jacent $S_t$, avec la même maturité finale $T_2 > T_1$ et le même prix d’exercice $K$.

\paragraph{Question :} 
Donner le prix de cette option à $t=0$ dans le modèle de Black--Scholes.    
\end{exerciseBox}


\subsection*{Solution :}
À la date $T_1$, la valeur de l’option chooser est
\[
V(T_1) = \max \Big( C_{T_1}(K,T_2), \; P_{T_1}(K,T_2) \Big),
\]
où $C_{T_1}(K,T_2)$ et $P_{T_1}(K,T_2)$ désignent respectivement le prix d’un call et d’un put européens de strike $K$ et échéance $T_2$, évalués à la date $T_1$.

D’après la parité call--put,
\[
C_{T_1}(K,T_2) - P_{T_1}(K,T_2) = S_{T_1} - K e^{-r(T_2-T_1)}.
\]

or 
\[
\max(x,y) = y + \max(x - y , 0)
\]

Ainsi,

\begin{align*}
\max(C_{T_1},P_{T_1}) &= P_{T_1}(K , T_2) + \max(C_{T_1}(K,T_2) - P_{T_1}(K,T_2),0)\\
&= P_{T_1}(K,T_2) + \max\Big(S_{T_1} - K e^{-r(T_2-T_1)} , 0 \Big) \\
&= P_{T_1}(K,T_2) + \max\Big(S_{T_1} - K' , 0 \Big) \quad avec \; \; K'= K e^{-r(T_2-T_1)}
\end{align*}


or 
$\max\Big(S_{T_1} - K e^{-r(T_2-T_1)} , 0 \Big)$ est le payoff d'une option call de strike $K' = K e^{-r(T_2-T_1)}$ et d'échéance $T_1$

Par conséquent, le prix de l’option chooser à $t=0$ s’écrit
\[
V(0) = P(0;K,T_2) + C_0\big(K e^{-r(T_2-T_1)},T_1\big),
\]
c’est-à-dire la somme :
\begin{itemize}
    \item d’un put européen de strike $K$ et maturité $T_2$, 
    \item plus d’un call européen de strike $K e^{-r(T_2-T_1)}$ et maturité $T_1$.
\end{itemize}



\section{Option d'échange avec devise}

\subsection*{Exercice :}

\begin{exerciseBox}[Option d'échange avec devise]
Soient $Z_t$ (actif en EUR), $Y_t$ (actif en USD) et $X_t$ (taux de change EUR/USD). \\ 
soit une option de payoff à maturité $T$ est 
\[
\Pi_T=(Z_T - X_T Y_T)^+.
\]
Dans le monde de Black–Scholes  donner le prix de l'option à t= 0 ? 
\end{exerciseBox}


\subsection*{Solution}

% On travaille dans Black--Scholes avec taux constants $r^{\EUR}$ (euro) et $r^{\USD}$ (dollar).
% On note $B_t^{\EUR}=e^{r^{\EUR} t}$ et $B_t^{\USD}=e^{r^{\USD} t}$.
% Soit $W_t=(W_t^{(1)},W_t^{(2)},W_t^{(3)})$ un mouvement brownien \emph{vectoriel} sous la mesure
% domestique $\,\mathbb{Q}^{\EUR}\,$, de matrice de corrélation
% $\rho_{ij}=\mathrm{Corr}(dW^{(i)}_t,dW^{(j)}_t)$.

% Sous $\mathbb{Q}^{\EUR}$, les dynamiques risque-neutres (drifts domestiques corrects) sont
% \begin{align*}
% \frac{dZ_t}{Z_t} &= r^{\EUR}\,dt + \sigma_Z\,dW_t^{(1)},\\
% \frac{dY_t}{Y_t} &= r^{\USD}\,dt + \sigma_Y\,dW_t^{(2)},\\
% \frac{dX_t}{X_t} &= (r^{\USD}-r^{\EUR})\,dt + \sigma_X\,dW_t^{(3)}.
% \end{align*}

Le payoff vaut $(Z_T - X_T Y_T)^+$ et le prix initial est
\begin{align*}
C_0
  &= \mathbb{E}^{\mathbb{Q}^{\EUR}}\!\Big[e^{-r^{\EUR}T}\,\big(Z_T - X_TY_T\big)^+\Big]\\
  &= \mathbb{E}^{\mathbb{Q}^{\EUR}}\!\Big[e^{-r^{\EUR}T} Z_T\,\mathbf{1}_{\{Z_T\ge X_TY_T\}}\Big]
   - \mathbb{E}^{\mathbb{Q}^{\EUR}}\!\Big[e^{-r^{\EUR}T} X_TY_T\,\mathbf{1}_{\{Z_T\ge X_TY_T\}}\Big].
\end{align*}


\begin{rappelBox}[Changement de numéraire]
\begin{itemize}
    \item  Soit $(\mathbb{Q}^{\alpha},\alpha_t)$ une mesure de probabilité et un numéraire de référence,
c'est-à-dire que pour tout actif $Z_t$ ne versant pas de dividendes,
le processus $\displaystyle \frac{Z_t}{\alpha_t}$ est une $\mathbb{Q}^{\alpha}$-martingale.

\item  Soit $(\beta_t)$ un \emph{numéraire simple}, c'est-à-dire un processus strictement positif
tel que $\displaystyle \big( \frac{\beta_t}{\alpha_t} \big)_t$  soit une $\mathbb{Q}^{\alpha}$-martingale.

\end{itemize}


Alors :

\begin{enumerate}[label=\roman*)]
\item Il existe une mesure $\mathbb{Q}^{\beta}$ équivalente à $\mathbb{Q}^{\alpha}$ telle que
\[
\left.\frac{d\mathbb{Q}^{\beta}}{d\mathbb{Q}^{\alpha}}\right|_{\mathcal{F}_t}
= \frac{\alpha_0}{\alpha_t}\,\frac{\beta_t}{\beta_0}.
\]
De plus, pour tout $F_T$ v.a,
\[
\mathbb{E}^{\mathbb{Q}^{\alpha}}\!\left[
\frac{\alpha_t}{\alpha_T} F_T \,\bigg|\, \mathcal{F}_t
\right]
=
\mathbb{E}^{\mathbb{Q}^{\beta}}\!\left[
\frac{\beta_t}{\beta_T} F_T \,\bigg|\, \mathcal{F}_t
\right].
\]

\item Si $\displaystyle \frac{Z_t}{\alpha_t}$ est une $\mathbb{Q}^{\alpha}$-martingale,
alors $\displaystyle \frac{Z_t}{\beta_t}$ est une $\mathbb{Q}^{\beta}$-martingale.

\end{enumerate}
\end{rappelBox}


\paragraph{Changement de numéraire $Z$.}
Définissons la mesure $\mathbb{Q}^{Z}$ par
\[
\frac{d\mathbb{Q}^{Z}}{d\mathbb{Q}^{\EUR}}\Big|_{\mathcal{F}_T}
=\frac{e^{-r^{\EUR}T}Z_T}{Z_0}.
\]
Alors
\begin{align*}
\mathbb{E}^{\mathbb{Q}^{\EUR}}\!\Big[e^{-r^{\EUR}T} Z_T\,\mathbf{1}_{\{Z_T\ge X_TY_T\}}\Big] 
&= \mathbb{E}^{\mathbb{Q}^Z}\!\Big[\frac{Z_0}{Z_T}Z_T \mathbf{1}_{\{Z_T\ge X_TY_T\}}\Big] \\
&= Z_0\mathbb{E}^{\mathbb{Q}^Z}\!\Big[\mathbf{1}_{\{Z_T\ge X_TY_T\}}\Big] \\
&= Z_0\,\mathbb{Q}^{Z}\!\big(Z_T\ge X_TY_T\big) \\
&= Z_0\,\mathbb{Q}^{Z}\!\big( \frac{X_TY_T}{Z_T} \le 1 \big) \\
% &= Z_0\,\mathcal{N}(d_1),
\end{align*}

\textbf{loi du} $\frac{X_TY_T}{Z_T}$ sous $\mathbb{Q}^Z$ :



Comme $Z_t$ est choisi comme numéraire, donc sous la  mesure $\mathbb{Q}^Z$ on  a que  $\big(\frac{X_t Y_t}{Z_t} \big)_t$ est une martingale.

Donc : 

\begin{align*}
d\!\left( \frac{X_t Y_t}{Z_t} \right)
&= \frac{X_t Y_t}{Z_t}
   \left[ \bigl(\sigma_X + \sigma_Y - \sigma_Z \bigr)\cdot dW_t^{Z} \right],
\end{align*}
où $W_t^{Z}$ est un mouvement brownien sous $\mathbb{Q}^Z$
et le drift est nul (martingale).

On en déduit que le processus est une exponentielle stochastique :
\begin{align*}
\frac{X_t Y_t}{Z_t}
&= \frac{X_0 Y_0}{Z_0}
   \exp\!\left\{
      \bigl(\sigma_X + \sigma_Y - \sigma_Z \bigr)\cdot W_t^{Z}
      - \tfrac12
        \bigl\|\sigma_X + \sigma_Y - \sigma_Z\bigr\|^2 t
   \right\}.
\end{align*}

Soit 
\[
\sigma := \sigma_X + \sigma_Y - \sigma_Z
\quad\text{et}\quad
\hat{\sigma} := \|\sigma\|
\]
% la volatilité effective (en norme euclidienne) du ratio
% \(\displaystyle \frac{X_t Y_t}{Z_t}\)
% sous la mesure \(\mathbb{Q}^Z\).
Alors
% \[
% \mathbb{Q}^Z\!\left(\frac{X_T Y_T}{Z_T}\le 1\right)
% = \Phi\!\left(
% \frac{ \ln\!\big(\tfrac{Z_0}{X_0 Y_0}\big)
%       + \tfrac12 \hat{\sigma}^2 T }
%      { \hat{\sigma}\sqrt{T} }
% \right),
% \]


\begin{align*}
\mathbb{Q}^{Z}\!\left(\frac{X_TY_T}{Z_T}\le 1\right)
&= \mathbb{Q}^{Z}\!\left(\ln\!\frac{X_TY_T}{Z_T}\le 0\right) \\
&= \Phi\!\left(\frac{\,0-\Big(\ln\!\frac{X_0Y_0}{Z_0}-\tfrac12\hat{\sigma}^2T\Big)}{\hat{\sigma}\sqrt{T}}\right)\quad \big( \text{on a} \; \sigma.W^Z_T = \hat{\sigma} \sqrt{T} G \; \; \text{avec} \; G \sim \mathcal{N}(0,1) \big) \\
&= \Phi\!\left(\frac{\ln\!\big(\tfrac{Z_0}{X_0Y_0}\big)+\tfrac12\hat{\sigma}^2T}{\hat{\sigma}\sqrt{T}}\right).
\end{align*}

où \(\Phi\) désigne la fonction de répartition de la loi normale standard \(\mathcal{N}(0,1)\).

Donc : 

\[
\boxed{
\mathbb{E}^{\mathbb{Q}^{\EUR}}\!\Big[e^{-r^{\EUR}T} Z_T\,\mathbf{1}_{\{Z_T\ge X_TY_T\}}\Big] = Z_0\Phi(d_1) \; avec \; \; d_1=\frac{\ln\!\big(\tfrac{Z_0}{X_0Y_0}\big)+\tfrac12\,\hat{\sigma}^2 T}{\hat{\sigma}\sqrt{T}}
}\]

% Ainsi, sous $\mathbb{Q}^Z$, le logarithme
% \(\ln \bigl(\tfrac{X_t Y_t}{Z_t}\bigr)\)
% suit une loi normale
% \[
% \mathcal{N}\!\left(
% \ln\!\bigl(\tfrac{X_0 Y_0}{Z_0}\bigr)
% - \tfrac12 \hat{\sigma}^2 t,\;
% \hat{\sigma}^2 t
% \right),
% \]
% où la volatilité ``d’échange'' est
% \[
% \hat{\sigma}^2
% = \sigma_X^2 + \sigma_Y^2 + \sigma_Z^2
% + 2\,\rho_{XY}\sigma_X\sigma_Y
% - 2\,\rho_{XZ}\sigma_X\sigma_Z
% - 2\,\rho_{YZ}\sigma_Y\sigma_Z.
% \]



% où $\mathcal{N}$ désigne la CDF normale standard et
% \[
% d_1=\frac{\ln\!\big(\tfrac{Z_0}{X_0Y_0}\big)+\tfrac12\,\hat{\sigma}^2 T}{\hat{\sigma}\sqrt{T}}.
% \]

\paragraph{Changement de numéraire $XY$.}
En prenant $XY$ comme numéraire,
\[
\frac{d\mathbb{Q}^{XY}}{d\mathbb{Q}^{\EUR}}\Big|_{\mathcal{F}_T}
=\frac{e^{-r^{\EUR}T}X_TY_T}{X_0Y_0},
\]
on obtient
\begin{align*}
\mathbb{E}^{\mathbb{Q}^{\EUR}}\!\Big[e^{-r^{\EUR}T} X_TY_T\,\mathbf{1}_{\{Z_T\ge X_TY_T\}}\Big]
= X_0Y_0\,\mathbb{Q}^{XY}\!\big(Z_T\ge X_TY_T\big)
= X_0Y_0\,\mathcal{N}(d_2),
\end{align*}
avec
\[
d_2 = d_1 - \hat{\sigma}\sqrt{T}.
\]


donc : 

\begin{align*}
\boxed{\;
C_0 \;=\; Z_0\,\mathcal{N}(d_1) \;-\; X_0Y_0\,\mathcal{N}(d_2),\qquad
d_1=\frac{\ln\!\big(\tfrac{Z_0}{X_0Y_0}\big)+\tfrac12\,\hat{\sigma}^2 T}{\hat{\sigma}\sqrt{T}},
\ \ d_2=d_1-\hat{\sigma}\sqrt{T}\; } 
\end{align*}


\section{Butterfly call}

\subsection*{Exercice :}

\begin{exerciseBox}[Butterfly call]
Soient trois calls européens de même maturité $T$ sur le même sous-jacent, aux strikes
% \[
% K_1=K-\Delta,\qquad K_2=K,\qquad K_3=K+\Delta,
% \]
\[
K_1=15,\qquad K_2=20,\qquad K_3=25,
\]

et de prix observés
\[
C(K_1)=9,\quad C(K_2)=6,\quad C(K_3)=2.
\]

Est-ce qu'il existe une opportunité d'arbitrage ?

\end{exerciseBox}

\subsection*{Solution :}


\paragraph{Convexité de $C(K,t)$ par rapport à $K$.}
On part de
\begin{align*}
C(K,t)
&= e^{-r(T-t)}\,\mathbb{E}^{\mathbb{Q}}\!\big[(S_T - K)_+ \,\big|\, \mathcal{F}_t\big].
\end{align*}
Sous des conditions usuelles d'interversion dérivée/espérance,
on dérive par rapport à $K$ :
\begin{align*}
\frac{\partial C}{\partial K}(K,t)
&= e^{-r(T-t)}\,\mathbb{E}^{\mathbb{Q}}\!\left[\frac{\partial}{\partial K}(S_T-K)_+ \,\Big|\, \mathcal{F}_t\right]
= e^{-r(T-t)}\,\mathbb{E}^{\mathbb{Q}}\!\big[-\mathbf{1}_{\{S_T>K\}} \,\big|\, \mathcal{F}_t\big] \\
&= -\,e^{-r(T-t)}\,\mathbb{Q}\!\big(S_T>K \,\big|\, \mathcal{F}_t\big).
\end{align*}
En dérivant à nouveau, on obtient :
\begin{align*}
\frac{\partial^2 C}{\partial K^2}(K,t)
&= e^{-r(T-t)}\, f_{S_T\mid \mathcal{F}_t}(K)
\;\;\ge\; 0,
\end{align*}
où $f_{S_T\mid \mathcal{F}_t}$ est la densité conditionnelle de $S_T$ donnée $\mathcal{F}_t$.



donc d'aprés l'inégalite de la convexite on a $\forall \; \lambda \in [0,1] \; \forall \; K_1 , K_2$ : 

\[
C(\lambda K_1 + (1- \lambda) K_2 ) \;\le\; \lambda C(K_1) + ( 1 - \lambda) C(K_2).
\]

ainis pour :
\[
\begin{cases}
K_1 = K - \Delta\\
K_2 = K \\
K_3 = K + \Delta \\
\lambda = \frac{1}{2}
\end{cases}
\]

On a : 
\[
C(K_2) \le \frac{1}{2} \big(C(K_1) + C(K_3) \big)
\]


\paragraph{Vérification numérique.}
En remplaçant par les données de marché :
\[
\frac{1}{2}\big[C(K_1)+C(K_3)\big]
= \frac{1}{2}(9+2) = 5.5.
\]
Or on observe
\[
C(K_2) = 6 \;>\; 5.5,
\]
ce qui viole l'inégalité de convexité.
Il existe donc une \textbf{opportunité d'arbitrage}.

\paragraph{3) Stratégie d'arbitrage.}
La violation de convexité suggère la stratégie dite de \emph{butterfly} :
\[
\text{Portefeuille} =
\text{long } 1 \text{ call strike } K_1
+\text{long } 1 \text{ call strike } K_3
-\text{short } 2 \text{ calls strike } K_2.
\]
\smallskip

\textbf{Coût initial.}  
Le coût à $t=0$ est
\[
\Pi_0 = C(K_1)+C(K_3)-2\,C(K_2)
= 9 + 2 - 2\times 6 = -1.
\]
Un coût négatif signifie que l'on \emph{encaisse} 1 euro à l'initiation.

\textbf{Payoff à maturité.}  
À l'échéance $T$, le payoff du portefeuille vaut
\begin{align*}
\Pi_T
&= \big(S_T - K_1\big)_+ + \big(S_T - K_3\big)_+
      - 2\big(S_T - K_2\big)_+ \\
&=
\begin{cases}
0, & S_T \le K_1,\\[0.4em]
S_T - K_1, & K_1 < S_T \le K_2,\\[0.4em]
(K_2-K_1) - (S_T - K_2)  \; \ge 0, & K_2 < S_T \le K_3,\\[0.4em]
(S_T-K_1) + (S_T-K_3) - 2(S_T - K_2) = 0, & S_T > K_3.
\end{cases}
\end{align*}
Ici $K_2-K_1=K_3-K_2=\Delta=5$, donc le payoff maximal vaut $\Delta=5$.

\textbf{Profit certain.}  
Comme $\Pi_T \ge 0$ pour tout $S_T$ et que l'on a reçu 1 euro initialement,
le profit à la maturité est toujours positif ou nul :
\[
\text{gain minimal} = 1
\]

\section{Put Spread}

\subsection*{Exercice :}

\begin{exerciseBox}[Put Spread]
On considère un sous-jacent $S$ et deux options put européennes de même maturité $T$ :  

 \begin{itemize}
     \item un put de strike $K + \varepsilon$, de prix $P_1 = 10$ ;  
    \item un put de strike $K$, de prix $P_2 = 11$.  
 \end{itemize}

On construit un \emph{put spread} en achetant le put de strike $K + \varepsilon$ et en vendant le put de strike $K$.  
\\
\textbf{Question} : \textit{y a-t-il une opportunité d’arbitrage ?}
\end{exerciseBox}



\subsection*{Solution :}


\textbf{Payoff du put spread :}
Le payoff d’un put de strike $K$ est $(K-S_T)^+$. Donc le payoff de la stratégie est
\[
\Pi_T \;=\; (K+\varepsilon - S_T)^+ \;-\; (K - S_T)^+ \;=\;
\begin{cases}
\varepsilon, & S_T < K,\\[3pt]
K+\varepsilon - S_T, & K \le S_T < K+\varepsilon,\\[3pt]
0, & S_T \ge K+\varepsilon.
\end{cases}
\]

\textbf{Le prix d'une option put européenne est croissant par rapport au strike }

Soit \( P(K) \) le prix, au temps \( t \), d'une option put de strike \( K \) et de maturité \( T \).
Sous la mesure risque-neutre, on a
\[
P(K) = e^{-r(T-t)} \, \mathbb{E}\big[ (K - S_T)_+ \,\big|\, \mathcal{F}_t \big].
\]
On dérive \( P(K) \) par rapport à \( K \) :
\[
\frac{\partial P}{\partial K}
= e^{-r(T-t)} \, \mathbb{E}\left[ 
\frac{\partial}{\partial K}(K - S_T)_+ \,\bigg|\, \mathcal{F}_t \right].
\]
Or, pour tout \( x \in \mathbb{R} \),
\[
\frac{\partial}{\partial K} (K - x)_+
= \mathbf{1}_{\{x < K\}},
\]
d'où
\[
\frac{\partial P}{\partial K}
= e^{-r(T-t)} \, \mathbb{E}\big[ \mathbf{1}_{\{S_T < K\}} \,\big|\, \mathcal{F}_t \big].
\]
En notant \( \mathbb{P}^t(\cdot) = \mathbb{P}(\cdot \mid \mathcal{F}_t) \), on obtient
\[
\frac{\partial P}{\partial K}
= e^{-r(T-t)} \, \mathbb{P}^t\big( S_T < K \big).
\]
Comme \( e^{-r(T-t)} > 0 \) et que la probabilité est toujours positive,
\[
\frac{\partial P}{\partial K} \ge 0.
\]
Ainsi, le prix d'une option put européenne est bien \textbf{croissant} par rapport au strike \( K \).


Donc, théoriquement  $P(K+\varepsilon) > P(K)$,Or, d'après l'énoncé, $P(K+\varepsilon) = 10$ et $P(K) = 11$,donc il y a une opportunité d'arbitrage.




\section{Parité call--put}

\subsection*{Exercice :}

\begin{exerciseBox}[Parité call--put]
On considère des options européennes (call $C_t(K,T)$ et put $P_t(K,T)$) de même strike $K$ et même maturité $T$,
écrites sur un sous-jacent $S_t$ ne versant pas de dividendes.
   

\medskip
\noindent
\textbf{Question.} Montrer la \emph{parité call--put} :
\[
C_t(K,T) - P_t(K,T) \;=\; S_t \;-\; K\,e^{-r(T-t)}.
\]

\end{exerciseBox}




\subsection*{Solution}

\textbf{Méthode 1 : Raisonnement d'arbitrage (réplication).}

Considérons deux portefeuilles :
\[
A := \begin{cases}
\text{long } 1 \text{ call } (K,T) \\
\text{long } K e^{-r(T-t)} \text{ d'obligation zéro-coupon}
\end{cases}
B :=  \begin{cases}
\text{long } 1 \text{ put } (K,T) \\
\text{long } 1 \text{ action } S
\end{cases}
\]
À l'échéance $T$, leurs payoffs valent
\begin{align*}
\Pi_T(A) &= (S_T - K)_+ + K \;=\; \max(S_T, K),\\
\Pi_T(B) &= (K - S_T)_+ + S_T \;=\; \max(S_T, K).
\end{align*}
Donc $\Pi_T(A)=\Pi_T(B)$ pour tout $S_T$. Par absence d'arbitrage, leurs valeurs aujourd'hui doivent coïncider :
\begin{align*}
C_t(K,T) + K e^{-r(T-t)} \;=\; P_t(K,T) + S_t
\;\;\Longleftrightarrow\;\;
C_t(K,T) - P_t(K,T) \;=\; S_t - K e^{-r(T-t)}.
\end{align*}

% \smallskip
% \noindent
% \textit{(Détection d'arbitrage en cas de violation.)}
% Si $C_t - P_t > S_t - K e^{-r(T-t)}$, alors le portefeuille $A$ est \emph{surévalué} par rapport à $B$ :
% on shorte $A$ et on achète $B$ pour encaisser un gain initial sans risque (les payoffs étant identiques à $T$).
% Le raisonnement s'inverse si $C_t - P_t < S_t - K e^{-r(T-t)}$.

\textbf{Méthode 2 : Par la mesure risque-neutre.}

Sous la mesure risque-neutre $\mathbb{Q}$,
\begin{align*}
C_t(K,T) &= e^{-r(T-t)}\,\mathbb{E}^{\mathbb{Q}}\!\big[(S_T - K)_+ \,\big|\, \mathcal{F}_t\big],\\
P_t(K,T) &= e^{-r(T-t)}\,\mathbb{E}^{\mathbb{Q}}\!\big[(K - S_T)_+ \,\big|\, \mathcal{F}_t\big].
\end{align*}
On utilise l'identité pointwise
\[
(S_T - K)_+ - (K - S_T)_+ = S_T - K,
\]
d’où
\begin{align*}
C_t(K,T) - P_t(K,T)
&= e^{-r(T-t)}\,\mathbb{E}^{\mathbb{Q}}\!\big[S_T - K \,\big|\, \mathcal{F}_t\big] \\
&= e^{-r(T-t)}\Big(\mathbb{E}^{\mathbb{Q}}[S_T \mid \mathcal{F}_t] - K\Big).
\end{align*}
Or, en absence de dividendes, $e^{-rt}S_t$ est une martingale sous $\mathbb{Q}$, donc
\[
\mathbb{E}^{\mathbb{Q}}[S_T \mid \mathcal{F}_t] = S_t\,e^{r(T-t)}.
\]
Par suite,
\begin{align*}
C_t(K,T) - P_t(K,T)
= e^{-r(T-t)}\big(S_t e^{r(T-t)} - K\big)
= S_t - K e^{-r(T-t)}.
\end{align*}

\paragraph{Remarque :}
Si le sous-jacent verse un dividende continu au taux $q$, on a :
\[
C_t - P_t \;=\; S_t\,e^{-q(T-t)} \;-\; K\,e^{-r(T-t)}.
\]

\section{Digtial Option : Dual Delta - Breeden-Litzenberger (1978)}

\subsection*{Exercice :}

\begin{exerciseBox}[Digtial Option :Dual Delta - Breeden-Litzenberger (1978)]
  On considère un actif risqué $(S_t)_{t \ge 0}$ sous la mesure risque-neutre $\mathbb{Q}$,
  évoluant selon le modèle de Black--Scholes avec taux sans risque constant $r$ et
  volatilité constante $\sigma > 0$ :
  \[
  dS_t = r S_t \, dt + \sigma S_t \, dW_t^{\mathbb{Q}}.
  \]
  
  On s'intéresse à une option digitale de type call, de maturité $T$ et de strike $K > 0$,
  de payoff
  \[
  \Phi(S_T) = \mathbf{1}_{\{S_T \ge K\}}.
  \]
  On note $D_t$ le prix à la date $t \le T$ de cette option.
  
  \begin{enumerate}
    \item[\textbf{Q1.}] Sous le modèle de Black--Scholes avec $r$ constant, donner une formule
    fermée pour $D_t$.
  
    \item[\textbf{Q2.}] On suppose maintenant que le taux court $(r_u)_{u\ge 0}$ est stochastique.
    Donner la formule de valorisation de $D_t$ sous la mesure risque-neutre ?
    % (b) Expliquer comment utiliser la mesure $T$-forward pour pricer $D_t$.
  
    \item[\textbf{Q3.}] On se place de nouveau dans le cas où $r$ est déterministe 
    \begin{enumerate}
      \item[(a)] Notons $C_t(K)$ le prix à la date $t$ d'une option call européenne de maturité $T$
      et de strike $K$. Exprimer $D_t(K)$ en fonction de $C_t(K)$ et du strike $K$.
  
      \item[(b)] On suppose que la volatilité implicite dépend du strike, i.e. $\sigma = \sigma(K)$,
      et que
      \[
        C_t(K) = f\big(K, \sigma(K)\big)
      \]
      pour une certaine fonction régulière $f$. Donner l'expression de $D_t(K)$ en fonction
      de $C_t$, de ses dérivées partielles, de $\sigma(K)$ et de la dérivée $\sigma'(K)$.
    \end{enumerate}
  \end{enumerate}
\end{exerciseBox}


\subsection*{Solution :}

\textbf{Q1. Formule fermée sous Black--Scholes avec $r$ constant.}

Sous la mesure risque-neutre $\mathbb{Q}$, le prix d'une option est donné par la formule
de valorisation par espérance actualisée :
\[
D_t = \mathbb{E}^{\mathbb{Q}}\!\left[ e^{-r (T-t)} \Phi(S_T) \mid \mathcal{F}_t \right]
     = e^{-r (T-t)} \, \mathbb{Q}\!\left( S_T \ge K \mid \mathcal{F}_t \right).
\]

Sous Black--Scholes, on sait que
\[
\ln S_T \mid \mathcal{F}_t \sim \mathcal{N}\Big(
\ln S_t + \big(r - \tfrac{1}{2}\sigma^2\big)(T-t), \; \sigma^2 (T-t)
\Big),
\]
ce qui conduit à
\[
\mathbb{Q}(S_T \ge K \mid \mathcal{F}_t)
= \mathbb{Q}\!\left( \ln S_T \ge \ln K \mid \mathcal{F}_t \right)
= N(d_2),
\]
où
\[
d_2 = \frac{\ln\big(\frac{S_t}{K}\big) + \big(r - \tfrac{1}{2}\sigma^2\big)(T-t)}
           {\sigma \sqrt{T-t}},
\]
et $N(\cdot)$ désigne la fonction de répartition d'une loi normale centrée réduite.

On obtient donc la formule fermée classique :
\[
\boxed{
D_t = e^{-r (T-t)} N(d_2).
}
\]

\bigskip

\textbf{Q2. Cas d'un taux court stochastique.}

On suppose maintenant que le taux court $(r_u)_{u \ge 0}$ est un processus stochastique
adapté à $(\mathcal{F}_u)_u$ et que l'on travaille sous la mesure risque-neutre $\mathbb{Q}$.

\medskip

\textbf{Formule sous la mesure risque-neutre.}

Le prix de l'option digitale est alors donné par
\[
D_t = \mathbb{E}^{\mathbb{Q}}\!\left[
\exp\!\left(-\int_t^T r_u \, du\right) \mathbf{1}_{\{S_T \ge K\}}
\;\Big|\; \mathcal{F}_t
\right].
\]

Cette formule est en général \emph{la} définition du prix, et l'obtention d'une forme
fermée dépend du choix de modèle conjoint pour $(S_t, r_t)$.

\medskip

\textbf{Utilisation de la mesure $T$-forward.}

On note $B(t,T)$ le prix à la date $t$ de la zero-coupon de maturité $T$ :
\[
B(t,T) =
\mathbb{E}^{\mathbb{Q}}\!\left[
\exp\!\left(-\int_t^T r_u \, du\right)
\;\Big|\; \mathcal{F}_t
\right].
\]

On introduit la mesure $T$-forward $\mathbb{Q}^T$ associée au numéraire $B(\cdot, T)$.
Par définition,
\[
\frac{d\mathbb{Q}^T}{d\mathbb{Q}}\Big|_{\mathcal{F}_T}
= \frac{\exp\!\left(-\int_0^T r_u \,du\right) B(t,T)}{B(0,T)},
\]

D'aprés le thoréme de Changement de numéraire on obtient alors la représentation suivante :
\[
D_t
= B(t,T)\, \mathbb{E}^{\mathbb{Q}^T}\!\left[
\mathbf{1}_{\{S_T \ge K\}} \mid \mathcal{F}_t
\right]
= B(t,T)\, \mathbb{Q}^T(S_T \ge K \mid \mathcal{F}_t).
\]

Autrement dit, sous la mesure $T$-forward, le prix de l'option digitale s'écrit
comme le produit du prix du zéro-coupon $B(t,T)$ par la probabilité (sous
$\mathbb{Q}^T$) que $S_T$ dépasse le strike $K$.

\bigskip

\textbf{Q3. Lien avec le prix de l'option call.}

On suppose de nouveau que le taux sans risque est déterministe (éventuellement non constant) :
\[
B_t = \exp\!\left(\int_0^t r_u \, du\right), \quad
B(t,T) = \exp\!\left(-\int_t^T r_u \, du\right).
\]

\medskip

\textbf{(a) Expression de $D_t(K)$ en fonction de $C_t(K)$.}

Le prix du call de strike $K$ et de maturité $T$ à la date $t$ s'écrit
\[
C_t(K) = \mathbb{E}^{\mathbb{Q}}\!\left[
B(t,T)\,(S_T - K)^+ \mid \mathcal{F}_t
\right]
= B(t,T)\, \mathbb{E}^{\mathbb{Q}}\!\left[(S_T - K)^+ \mid \mathcal{F}_t \right].
\]

En termes de densité de $S_T$,

\[
C_t(K) = B(t,T)\int_0^{+\infty} (s-K)_+ f_{S_T\mid\mathcal{F}_t}(s)\, ds.
\]

On différentie formellement $C_t(K)$ par rapport à $K$ (en supposant les régularités
adéquates) :
\[
\frac{\partial C_t}{\partial K}(K)
= B(t,T)\, \frac{\partial}{\partial K}
\left( \int_0^{+\infty} (s-K)_+ f_{S_T\mid\mathcal{F}_t}(s)\, ds \right).
\]

or 
\[
\frac{\partial (s - K)_+}{\partial K} = - \mathbf{1}_{s \ge K}
\]

En dérivant sous le signe intégral, on obtient
\[
\frac{\partial C_t}{\partial K}(K)
= B(t,T)\left(
- \int_0^{+\infty} \mathbf{1}_{s \ge K} f_{S_T\mid\mathcal{F}_t}(s)\, ds
\right)
= -B(t,T)\, \mathbb{Q}(S_T \ge K \mid \mathcal{F}_t).
\]

Or, on a exactement
\[
D_t(K) = B(t,T)\, \mathbb{Q}(S_T \ge K \mid \mathcal{F}_t).
\]

On en déduit la relation fondamentale :
\[
\boxed{
D_t(K) = - \frac{\partial C_t}{\partial K}(K).
}
\]

\medskip

\textbf{(b) Cas d'une volatilité implicite dépendant du strike.}

On suppose maintenant que le prix du call est donné par
\[
C_t(K) = f\big(K, \sigma(K)\big),
\]
où $\sigma(K)$ est la volatilité implicite associée au strike $K$ et $f$ une fonction
régulière en ses arguments .

Le prix $C_t(K)$ est maintenant une fonction composée de $K$ via deux canaux :
le strike lui-même et la volatilité implicite $\sigma(K)$. La dérivée totale de $C_t$
par rapport à $K$ s'écrit alors
\[
\frac{d C_t}{d K}(K)
= \frac{\partial f}{\partial K}\big(K, \sigma(K)\big)
  + \frac{\partial f}{\partial \sigma}\big(K, \sigma(K)\big)\, \sigma'(K).
\]

En utilisant la relation de la question (a),
\[
D_t(K) = - \frac{d C_t}{d K}(K),
\]
on obtient
\[
\boxed{
D_t(K)
= -\left[
\frac{\partial f}{\partial K}\big(K, \sigma(K)\big)
+ \frac{\partial f}{\partial \sigma}\big(K, \sigma(K)\big)\, \sigma'(K)
\right].
}
\]

Dans le cas où $f$ est la formule de Black--Scholes,
\[
\frac{\partial f}{\partial \sigma} = \text{Vega}(K,\sigma) ,
\]
et la dernière expression peut se réécrire en termes de dérivée partielle par rapport
au strike (avec $\sigma$ fixée) et de la Vega, corrigée par la pente de la surface
de volatilité implicite $\sigma'(K)$ :
\[
D_t(K) = -\left[ C_K^{\text{BS}}(K,\sigma(K)) + \text{Vega}(K,\sigma(K))\, \sigma'(K) \right].
\]


\section{Call européen avec sous-jacent gaussien}

\subsection*{Exercice :}

\begin{exerciseBox}[Call européen avec sous-jacent gaussien]
Suppose that the stock price at maturity \(T\) is normally distributed:
\[
S_T \sim \mathcal{N}(S_0,\delta^2).
\]
What is the expected value (present value under the pricing measure) of an \emph{at-the-money} European call option expiring at time \(T\), i.e. with strike \(K=S_0\)?
\end{exerciseBox}



\subsection*{Solution :}

Le prix (sous la mesure de pricing) d’un call européen est
\[
c_0 \;=\; e^{-rT}\,\mathbb{E}\!\left[(S_T-K)_+\right].
\]
Comme \(S_T \sim \mathcal{N}(S_0,\delta^2)\), on peut écrire
\[
S_T = S_0 + \delta Z,
\qquad Z \sim \mathcal{N}(0,1).
\]
À la monnaie, \(K=S_0\), donc
\[
(S_T-K)_+ = (S_0+\delta Z - S_0)_+ = (\delta Z)_+ = \delta\, Z\,\mathbf{1}_{\{Z\ge 0\}}.
\]
Ainsi
\[
c_0
= e^{-rT}\,\delta\,\mathbb{E}\!\left[ Z\,\mathbf{1}_{\{Z\ge 0\}}\right].
\]
Calculons \(\mathbb{E}[ Z\,\mathbf{1}_{\{Z\ge 0\}}]\) pour \(Z\sim\mathcal{N}(0,1)\) :
\[
\mathbb{E}\!\left[ Z\,\mathbf{1}_{\{Z\ge 0\}}\right]
= \int_0^{+\infty} x \frac{1}{\sqrt{2\pi}} e^{-x^2/2}\,dx.
\]
Par le changement \(u = x^2/2\) (donc \(du = x\,dx\)),
\[
\int_0^{+\infty} x \frac{1}{\sqrt{2\pi}} e^{-x^2/2}\,dx
= \frac{1}{\sqrt{2\pi}}\int_0^{+\infty} e^{-u}\,du
= \frac{1}{\sqrt{2\pi}}.
\]
Donc
\[
\boxed{\;\mathbb{E}\!\left[(S_T-S_0)_+\right] = \frac{\delta}{\sqrt{2\pi}}\;}
\qquad\text{et}\qquad
\boxed{\;c_0 = e^{-rT}\,\frac{\delta}{\sqrt{2\pi}}\;}.
\]

\section{Call option at-the-money}

\subsection*{Exercice :}

\begin{exerciseBox}[Call option at-the-money]
Approximately how much would a one-month at-the-money European call option be worth if:
\begin{itemize}
  \item the spot price is \(S_t = 1\),
  \item the notional is \$1{,}000{,}000,
  \item the maturity is one month (\(\tau = T-t = \tfrac{1}{12}\)),
  \item the volatility is \(\sigma = 10\%\),
  \item the interest rate is \(r=0\).
\end{itemize}
\end{exerciseBox}

\subsection*{Solution :}

Under the Black--Scholes model, the price of a European call option is
\[
C_t = S_t \Phi(d_1) - K e^{-r\tau}\Phi(d_2),
\]
where
\[
d_1 = \frac{\ln(S_t/K) + \left(r + \tfrac12\sigma^2\right)\tau}{\sigma\sqrt{\tau}},
\qquad
d_2 = d_1 - \sigma\sqrt{\tau}.
\]

At-the-money, \(S_t = K\), so \(\ln(S_t/K)=0\). With \(r=0\), this gives
\[
d_1 = \frac{\tfrac12\sigma^2\tau}{\sigma\sqrt{\tau}}
= \frac12\sigma\sqrt{\tau},
\qquad
d_2 = -\frac12\sigma\sqrt{\tau}.
\]

Hence,
\[
C_t = S_t\Big(\Phi(\tfrac12\sigma\sqrt{\tau}) - \Phi(-\tfrac12\sigma\sqrt{\tau})\Big).
\]
Using the symmetry property \(\Phi(-x)=1-\Phi(x)\),
\[
C_t = S_t\Big(2\Phi(\tfrac12\sigma\sqrt{\tau}) - 1\Big).
\]

For small \(x\), the standard normal CDF can be approximated by
\[
\Phi(x) \approx \frac12 + \frac{x}{\sqrt{2\pi}}.
\]
Therefore,
\[
\Phi\!\left(\tfrac12\sigma\sqrt{\tau}\right)
\approx \frac12 + \frac{\sigma\sqrt{\tau}}{2\sqrt{2\pi}}.
\]

Substituting into the call price,
\[
C_t \approx S_t \left(2\left(\frac12 + \frac{\sigma\sqrt{\tau}}{2\sqrt{2\pi}}\right) - 1\right)
= \frac{\sigma}{\sqrt{2\pi}} S_t \sqrt{\tau}.
\]

Numerically,
\[
\frac{1}{\sqrt{2\pi}} \approx 0.40,
\]
so
\[
\boxed{
C_t \approx 0.4\, S_t\, \sigma \sqrt{\tau}.
}
\]

With \(S_t=1\), \(\sigma=0.1\), and \(\tau=\tfrac{1}{12}\),
\[
C_t \approx 0.4 \times 0.1 \times \sqrt{\tfrac{1}{12}}
\approx 0.0115.
\]

For a notional of \$1{,}000{,}000, the option value is therefore
\[
\boxed{C_t \approx \$11{,}500.}
\]

\section{FX Forward}

\subsection*{Exercice :}

\begin{exerciseBox}[FX Forward et parité des taux]
On considère le taux de change \emph{spot} à la date $t=0$ noté $S_0$, défini par
\[
S_0 = \text{prix de } 1~\mathrm{USD} \text{ exprimé en EUR}
\qquad\Longleftrightarrow\qquad
1~\mathrm{USD} = S_0~\mathrm{EUR}.
\]
On note $r_{\$}$ le taux sans risque (sur $[0,T]$) en USD et $r_{\euro}$ le taux sans risque (sur $[0,T]$) en EUR.
On suppose qu'il n'y a pas d'opportunité d'arbitrage.

\medskip
\noindent\textbf{Q1.} Qu'est-ce qu'un \emph{FX forward} ?

\smallskip
\noindent\textbf{Q2.} En utilisant une stratégie d'arbitrage (réplication), démontrer la formule de pricing du taux forward $F_{0,T}$.

\end{exerciseBox}

\subsection*{Solution :}

\subsection*{Réponse Q1 : définition d'un FX forward}
Un \emph{FX forward} est un contrat qui fixe à la date $t=0$ un taux $F_{0,T}$ auquel deux contreparties
s'engagent à échanger, à la date future $T$, un montant notionnel d'une devise contre un montant équivalent
dans l'autre devise.
Ici, avec la convention $S_0$ en \emph{EUR par USD}, le contrat peut s'écrire :
\[
\text{à } T,\ \text{échanger } 1~\mathrm{USD} \ \text{contre } F_{0,T}~\mathrm{EUR}.
\]

\subsection*{Réponse Q2 : démonstration (absence d'arbitrage)}
On cherche le \emph{fair forward} $F_{0,T}$ compatible avec l'absence d'arbitrage.

\medskip
\noindent\textbf{Stratégie (cash-and-carry) à $t=0$:}
\begin{itemize}
  \item Emprunter $S_0$ EUR au taux $r_{\euro}$.
  \item Convertir $S_0$ EUR en $1$ USD au spot (car $1~\mathrm{USD}=S_0~\mathrm{EUR}$).
  \item Placer $1$ USD au taux $r_{\$}$ jusqu'à $T$.
\end{itemize}
La valeur initiale du portefeuille est nulle (on emprunte exactement ce qu'on convertit) :
\[
V_0 = +1~\mathrm{USD} - S_0~\mathrm{EUR} = 0.
\]

\medskip
\noindent\textbf{Valeur à l'échéance $T$:}
\begin{itemize}
  \item Le placement USD devient $(1+r_{\$})$ USD.
  \item La dette EUR devient $S_0(1+r_{\euro})$ EUR.
\end{itemize}

Pour éliminer le risque de change, on convertit les USD en EUR \emph{au taux forward} $F_{0,T}$ (EUR par USD).
La valeur certaine en EUR à $T$ vaut alors :
\[
V_T^{(\mathrm{EUR})} = (1+r_{\$})\,F_{0,T} - S_0(1+r_{\euro}).
\]

\medskip
\noindent\textbf{Absence d'arbitrage :} un portefeuille de coût initial nul ne peut pas générer un gain certain,
donc il faut imposer
\[
V_T^{(\mathrm{EUR})}=0.
\]
Ainsi,
\[
(1+r_{\$})\,F_{0,T} - S_0(1+r_{\euro}) = 0
\qquad\Longrightarrow\qquad
\boxed{
F_{0,T} = S_0\,\frac{1+r_{\euro}}{1+r_{\$}} }.
\]

\subsection*{Conclusion :}
On définit le taux de change $X_t^{(\text{USD/EUR})}$ par
\[
1~\text{USD} = X_t^{(\text{USD/EUR})}~\text{EUR}.
\]
Le taux forward à maturité $T$ est donné par
\[
\boxed{
X_T^{(\text{USD/EUR})}
=
X_0^{(\text{USD/EUR})}
\frac{1+r_{\text{EUR}}}{1+r_{\text{USD}}}
}
\]

\[
\boxed{
X_T^{(\text{USD/EUR})}
=
X_0^{(\text{USD/EUR})}
\exp\!\left(
\bigl(r_{\text{EUR}} - r_{\text{USD}}\bigr)T
\right)
}
\]


\section{option digitale multi-actifs}



\subsection*{Exercice : }

\begin{exerciseBox}[option digitale multi-actifs]
On considère deux actifs financiers $S_1$ et $S_2$ évoluant sous la mesure risque-neutre $\mathbb{Q}$ selon des dynamiques de type Black--Scholes :
\[
\frac{dS_i(t)}{S_i(t)} = (r - q_i)\,dt + \sigma_i\,dW_i(t),
\qquad i = 1,2,
\]
où :
\begin{itemize}
  \item $r$ est le taux sans risque constant,
  \item $q_i$ est le taux de dividende de l’actif $S_i$,
  \item $\sigma_i > 0$ est la volatilité de $S_i$,
  \item $(W_1, W_2)$ est un mouvement brownien bidimensionnel tel que
  \[
  \langle W_1, W_2 \rangle_t = \rho t, \qquad \rho \in (-1,1).
  \]
\end{itemize}

On considère une option digitale de maturité $T$ dont le payoff est donné par :
\[
\mathbf{1}_{\{ S_1(T) \ge S_2(T) \}}.
\]

\medskip

\noindent
\textbf{Question :} Déterminer la formule fermée du prix de cette option à la date $t=0$.
\end{exerciseBox}

\subsection*{Solution :}

Sous la mesure risque-neutre, le prix de l’option à la date $0$ est donné par :
\[
V_0 = e^{-rT} \mathbb{E}^{\mathbb{Q}}\!\left[ \mathbf{1}_{\{ S_1(T) \ge S_2(T) \}} \right]
= e^{-rT} \mathbb{P}^{\mathbb{Q}}\!\left( S_1(T) \ge S_2(T) \right).
\]

\subsection*{Étude du ratio $\boldsymbol{S_1(T)/S_2(T)}$}

On introduit la variable aléatoire :
\[
X := \ln\!\left( \frac{S_1(T)}{S_2(T)} \right).
\]

À partir des dynamiques de $S_1$ et $S_2$, on obtient :
\begin{align*}
X
&= \ln\!\left( \frac{S_1(0)}{S_2(0)} \right)
+ \left[(r - q_1 - \tfrac12 \sigma_1^2) - (r - q_2 - \tfrac12 \sigma_2^2)\right] T \\
&\quad + \sigma_1 W_1(T) - \sigma_2 W_2(T).
\end{align*}

La variable $X$ suit donc une loi normale :
\[
X \sim \mathcal{N}(\mu_X, \sigma_X^2),
\]
avec :
\[
\mu_X
= \ln\!\left( \frac{S_1(0)}{S_2(0)} \right)
+ \left(q_2 - q_1 - \tfrac12 \sigma_1^2 + \tfrac12 \sigma_2^2 \right) T,
\]

\paragraph{Calcul de la variance de $\sigma_1 W_1(T)-\sigma_2 W_2(T)$.}
Posons
\[
Z := \sigma_1 W_1(T)-\sigma_2 W_2(T).
\]
En utilisant la formule $\mathrm{Var}(X+Y)=\mathrm{Var}(X)+\mathrm{Var}(Y)+2\,\mathrm{Cov}(X,Y)$ (et $\mathrm{Var}(-Y)=\mathrm{Var}(Y)$), on obtient
\begin{align*}
\mathrm{Var}(Z)
&= \sigma_1^2\,\mathrm{Var}(W_1(T))+\sigma_2^2\,\mathrm{Var}(W_2(T))
-2\sigma_1\sigma_2\,\mathrm{Cov}\!\big(W_1(T),W_2(T)\big).
\end{align*}
Comme $W_1$ et $W_2$ sont des browniens, $\mathbb{E}[W_i(T)]=0$ et $\mathrm{Var}(W_i(T))=\mathbb{E}[W_i(T)^2]=T$.

Il reste à calculer $\mathbb{E}[W_1(T)W_2(T)]$.
On utilise la propriété suivante : si $M$ et $N$ sont deux martingales (càdlàg) nulles en $0$ et de carré intégrable, alors
\[
MN-\langle M,N\rangle
\quad \text{est une martingale.}
\]
En particulier, avec $M=W_1$ et $N=W_2$,
\[
W_1(t)W_2(t)-\langle W_1,W_2\rangle_t
\quad \text{est une martingale.}
\]
En prenant l'espérance à l'instant $T$ et en utilisant $W_1(0)W_2(0)-\langle W_1,W_2\rangle_0=0$, on obtient
\[
\mathbb{E}\!\left[W_1(T)W_2(T)-\langle W_1,W_2\rangle_T\right]=0
\quad\Longrightarrow\quad
\mathbb{E}[W_1(T)W_2(T)]=\mathbb{E}\!\left[\langle W_1,W_2\rangle_T\right].
\]
Or, par hypothèse $\langle W_1,W_2\rangle_t=\rho t$, donc
\[
\mathbb{E}[W_1(T)W_2(T)]=\rho T,
\qquad
\mathrm{Cov}(W_1(T),W_2(T))=\rho T.
\]
Finalement,
\[
\mathrm{Var}(Z)
=\sigma_1^2 T+\sigma_2^2 T-2\sigma_1\sigma_2\,\rho T
= T\big(\sigma_1^2+\sigma_2^2-2\rho\sigma_1\sigma_2\big).
\]

Donc :

\[
\boxed{
\sigma_X^2
= T \left( \sigma_1^2 + \sigma_2^2 - 2 \rho \sigma_1 \sigma_2 \right).
}
\]



\subsection*{Calcul de la probabilité}

On a :
\[
\mathbb{P}^{\mathbb{Q}}\!\left( S_1(T) \ge S_2(T) \right)
= \mathbb{P}(X \ge 0)
= \Phi\!\left( \frac{\mu_X}{\sigma_X} \right),
\]
où $\Phi$ désigne la fonction de répartition de la loi normale centrée réduite.

\subsection*{Prix de l’option}

Finalement, le prix de l’option digitale est donné par :
\[
\boxed{
V_0
= e^{-rT}
\Phi\!\left(
\frac{
\ln\!\left( \frac{S_1(0)}{S_2(0)} \right)
+ \left(q_2 - q_1 - \tfrac12 \sigma_1^2 + \tfrac12 \sigma_2^2 \right) T
}{
\sqrt{
T \left( \sigma_1^2 + \sigma_2^2 - 2 \rho \sigma_1 \sigma_2 \right)
}
}
\right)
}
\]

\medskip

\noindent
Dans le cas particulier sans dividendes ($q_1 = q_2 = 0$), la formule se simplifie en :
\[
V_0
= e^{-rT}
\Phi\!\left(
\frac{
\ln\!\left( \frac{S_1(0)}{S_2(0)} \right)
+ \tfrac12 (\sigma_2^2 - \sigma_1^2) T
}{
\sqrt{
T \left( \sigma_1^2 + \sigma_2^2 - 2 \rho \sigma_1 \sigma_2 \right)
}
}
\right).
\]



\section{Option Margrabe}


\subsection*{Exercice :}

\begin{exerciseBox}[Option Margrabe]

On considére deux actifs $S_1$ et $S_2$ définis sur un espace probabilise
$(\Omega,\mathcal{F},(\mathcal{F}_t)_{t\ge 0},\mathbb{Q})$.
Sous la mesure risque-neutre $\mathbb{Q}$ associ\'ee au num\'eraire mon\'etaire
$B_t=e^{rt}$, les dynamiques sont de type Black--Scholes :
\[
\frac{dS_i(t)}{S_i(t)}=(r-q_i)\,dt+\sigma_i\,dW_i(t),\qquad i=1,2,
\]
o\`u $r$ est constant, $q_i$ est le taux de dividende continu de $S_i$,
$\sigma_i>0$, et $(W_1,W_2)$ est un brownien bidimensionnel tel que
\[
\langle W_1,W_2\rangle_t=\rho t,\qquad \rho\in(-1,1).
\]

On \'{e}tudie l'option d'\'echange (option de Margrabe) de maturit\'e $T$ et de payoff
\[
\big(S_1(T)-S_2(T)\big)^+.
\]

\medskip
\noindent
\textbf{Question :} Donner une formule fermée pour le prix $V_0$ de cette option à la date $0$ ?
\end{exerciseBox}

\subsection*{Solution :}



\subsection*{a. Modèle}

On considère deux actifs risqués $(S_t^1)_{t \ge 0}$ et $(S_t^2)_{t \ge 0}$ évoluant sous la mesure risque-neutre $\mathbb Q$ selon les dynamiques de Black--Scholes avec dividendes continus :
\[
\begin{cases}
\dfrac{dS_t^1}{S_t^1} = (r - q_1)\,dt + \sigma_1\, dW_t^1, \\
\dfrac{dS_t^2}{S_t^2} = (r - q_2)\,dt + \sigma_2\, dW_t^2,
\end{cases}
\]
où
\[
d\langle W^1, W^2 \rangle_t = \rho\,dt,
\]
$r$ est le taux sans risque, $q_1, q_2$ les taux de dividendes, $\sigma_1, \sigma_2$ les volatilités et $\rho \in [-1,1]$ le coefficient de corrélation.

Le numéraire de référence est le compte bancaire
\[
B_t = e^{rt}.
\]

\subsection*{b. Payoff}

On considère une option d’échange (option de Margrabe) de maturité $T$, de payoff
\[
\Phi_T = (S_T^1 - S_T^2)_+.
\]

Son prix à la date $0$ est
\[
C_0 = \mathbb E^{\mathbb Q}\left[ e^{-rT}\,(S_T^1 - S_T^2)_+ \right].
\]

\subsection*{c. Changement de numéraire}

On introduit le numéraire associé au second actif avec dividendes réinvestis :
\[
N_t := e^{q_2 t} S_t^2.
\]
Le processus $\left( \frac{N_t}{B_t} \right)_{t \ge 0}$ est une $\mathbb Q$-martingale.

On définit alors la mesure $\mathbb Q^{(2)}$ associée au numéraire $N_t$ par
\[
\left.\frac{d\mathbb Q^{(2)}}{d\mathbb Q}\right|_{\mathcal F_t}
= \frac{N_t / B_t}{N_0 / B_0}.
\]

Par le théorème du changement de numéraire, pour toute variable aléatoire intégrable $H_T$,
\[
\mathbb E^{\mathbb Q}\left[ \frac{B_0}{B_T} H_T \right]
= \mathbb E^{\mathbb Q^{(2)}}\left[ \frac{N_0}{N_T} H_T \right].
\]

En particulier,
\[
C_0
= \mathbb E^{\mathbb Q^{(2)}}\left[ \frac{N_0}{N_T} (S_T^1 - S_T^2)_+ \right]
= S_2(0)\,\mathbb E^{\mathbb Q^{(2)}}\left[ \left( \frac{S_T^1}{S_T^2} - 1 \right)_+ \right].
\]


\subsection*{d. Dynamique du ratio sous la mesure $\mathbb Q^{(2)}$}

On rappelle que la mesure $\mathbb Q^{(2)}$ est définie par le changement de numéraire
\[
N_t = e^{q_2 t} S_t^2.
\]
La densité de Radon--Nikodym associée est donnée par
\[
\left.\frac{d\mathbb Q^{(2)}}{d\mathbb Q}\right|_{\mathcal F_t}
= \frac{N_t / B_t}{N_0 / B_0}
= \frac{e^{q_2 t} S_t^2}{e^{rt}} \cdot \frac{1}{S_2(0)}.
\]

Sous la mesure $\mathbb Q$, le processus $S_t^2$ s’écrit explicitement :
\[
S_t^2 = S_2(0)\,
\exp\!\left(
\left(r - q_2 - \tfrac{1}{2}\sigma_2^2\right)t
+ \sigma_2 W_t^2
\right).
\]
En injectant cette expression dans la densité, on obtient :
\[
\left.\frac{d\mathbb Q^{(2)}}{d\mathbb Q}\right|_{\mathcal F_t}
=
\exp\!\left(
\sigma_2 W_t^2 - \tfrac{1}{2}\sigma_2^2 t
\right).
\]

Cette densité est l’exponentielle de Doléans-Dade associée au processus
\[
\int_0^t \sigma_2\, dW_s^2.
\]
Par le théorème de Girsanov, il existe alors un mouvement brownien
$(W_t^{2,\mathbb Q^{(2)}})_{t \ge 0}$ sous la mesure $\mathbb Q^{(2)}$ tel que
\[
\boxed{
dW_t^{2,\mathbb Q^{(2)}} = dW_t^2 - \sigma_2\,dt.
}
\]

\medskip

Les mouvements browniens $W^1$ et $W^2$ étant corrélés sous $\mathbb Q$ avec
\[
d\langle W^1, W^2 \rangle_t = \rho\,dt,
\]
on introduit la décomposition orthogonale
\[
W_t^1 = \rho W_t^2 + \sqrt{1-\rho^2}\, Z_t,
\]
où $(Z_t)_{t \ge 0}$ est un mouvement brownien indépendant de $W^2$ sous $\mathbb Q$.

Comme la densité du changement de mesure dépend uniquement de $W^2$, le processus
$Z_t$ reste un mouvement brownien sous $\mathbb Q^{(2)}$. On en déduit que
\[
\begin{aligned}
dW_t^{1,\mathbb Q^{(2)}}
&= \rho\, dW_t^{2,\mathbb Q^{(2)}} + \sqrt{1-\rho^2}\, dZ_t \\
&= \rho(dW_t^2 - \sigma_2 dt) + \sqrt{1-\rho^2}\, dZ_t \\
&= dW_t^1 - \rho\sigma_2\,dt.
\end{aligned}
\]

On obtient ainsi la relation entre les browniens sous les deux mesures :
\[
\boxed{
\begin{aligned}
dW_t^{2,\mathbb Q^{(2)}} &= dW_t^2 - \sigma_2\,dt, \\
dW_t^{1,\mathbb Q^{(2)}} &= dW_t^1 - \rho\sigma_2\,dt,
\end{aligned}
}
\]
avec
\[
d\langle W^{1,\mathbb Q^{(2)}}, W^{2,\mathbb Q^{(2)}} \rangle_t = \rho\,dt.
\]

Donc :
\[
\boxed{
\begin{cases}
\dfrac{dS_t^1}{S_t^1} = (r - q_1 + \rho \sigma_1 \sigma_2)\,dt + \sigma_1\,dW^{1,\mathbb Q^{(2)}}, \\
\dfrac{dS_t^2}{S_t^2} = (r - q_2 + \sigma_2^2)\,dt + \sigma_2\, dW^{2,\mathbb Q^{(2)}},  \\
d\langle W^{1,\mathbb Q^{(2)}}, W^{2,\mathbb Q^{(2)}} \rangle_t = \rho\,dt,
\end{cases}
}
\]


\medskip

En appliquant la formule d’Itô au processus
\[
R_t = \frac{S_t^1}{S_t^2},
\]
et en utilisant les dynamiques sous $\mathbb Q^{(2)}$, on obtient :
\[
\frac{dR_t}{R_t}
= (q_2 - q_1)\,dt
+ \sigma_1\, dW_t^{1,\mathbb Q^{(2)}}
- \sigma_2\, dW_t^{2,\mathbb Q^{(2)}}.
\]

Le terme de diffusion peut être regroupé en un unique brownien :
\[
\frac{dR_t}{R_t}
= (q_2 - q_1)\,dt
+ \sigma_{\text{eff}}\, dW_t^{\mathbb Q^{(2)}},
\]
où
\[
\sigma_{\text{eff}}
= \sqrt{\sigma_1^2 + \sigma_2^2 - 2\rho\sigma_1\sigma_2}.
\]



\subsection*{e. Pricing}

Sous la mesure $\mathbb Q^{(2)}$, le processus $R_t$ suit une dynamique lognormale.
L’option $(R_T - 1)_+$ est donc une option de type Call de strike $1$.

En appliquant la formule de Black--Scholes, on obtient :
\[
\mathbb E^{\mathbb Q^{(2)}}\left[ (R_T - 1)_+ \right]
= R_0 e^{- q_1T} N(d_1) - e^{- q_2T}N(d_2),
\]
avec
\[
d_1 = \frac{\ln(R_0) + (q_2 - q_1 + \tfrac{1}{2}\sigma_{\text{eff}}^2)T}
{\sigma_{\text{eff}}\sqrt{T}},
\qquad
d_2 = d_1 - \sigma_{\text{eff}}\sqrt{T}.
\]

\subsection*{f. Formule finale de Margrabe}

En utilisant $R_0 = \dfrac{S_1(0)}{S_2(0)}$, on obtient finalement :
\[
\boxed{
C_0
= S_1(0)e^{-q_1 T} N(d_1)
- S_2(0)e^{-q_2 T} N(d_2)
}
\]
avec
\[
\boxed{
\begin{aligned}
d_1 &= \frac{\ln\!\left( \dfrac{S_1(0)}{S_2(0)} \right)
+ \left(q_2 - q_1 + \tfrac{1}{2}\sigma_{\text{eff}}^2\right)T}
{\sigma_{\text{eff}}\sqrt{T}},\\[6pt]
d_2 &= d_1 - \sigma_{\text{eff}}\sqrt{T},\\[6pt]
\sigma_{\text{eff}} &= \sqrt{\sigma_1^2 + \sigma_2^2 - 2\rho\sigma_1\sigma_2}.
\end{aligned}
}
\]

\medskip
Cette formule est connue sous le nom de \emph{formule de Margrabe}.


\section{Valeur intrinsèque et exercice anticipé d'un call américain}

\begin{exerciseBox}[Valeur intrinsèque et exercice anticipé d'un call américain]

\textbf{Énoncé.}
\begin{enumerate}
\item[(Q1)] Montrer que, pour tout $t \in [0,T]$, le prix d'un call européen de strike $K$ et maturité $T$ vérifie
\[
C_t \;\ge\; (S_t - K)_+.
\]
\item[(Q2)] Justifier que, pour un \emph{call américain} sur une action ne versant \emph{aucun dividende} ($q=0$), il n'est jamais optimal d'exercer avant l'échéance : l'exercice optimal a lieu à $T$.
\end{enumerate}
\end{exerciseBox}


\subsection*{Solution :}

On suppose que l'actif sous-jacent $(S_t)_{t\in[0,T]}$ est modélisé sous la mesure risque-neutre $\mathbb Q$ (numéraire $B_t=e^{rt}$), et que le taux sans risque $r$ est constant.

%------------------------
\subsection*{Q1. Le call est toujours supérieur à sa valeur intrinsèque}

Considérons un call européen de payoff $(S_T-K)_+$.  
Construisons le portefeuille suivant à la date $t$ :
\[
\Pi_t \;:=\; C_t \;+\; K\,B(t,T),
\]
où $B(t,T)=e^{-r(T-t)}$ désigne le prix à $t$ d'un zéro-coupon qui paie $1$ à $T$.

À l'échéance, la valeur de ce portefeuille vaut :
\[
\Pi_T \;=\; C_T + K \;=\; (S_T-K)_+ + K \;=\; \max(S_T,K).
\]
Or, pour tout $x\ge 0$, on a $\max(x,K)\ge x$, donc
\[
\Pi_T \;=\; \max(S_T,K) \;\ge\; S_T.
\]

Le portefeuille $\Pi$ domine donc le payoff $S_T$ à l'échéance.  
Par absence d'arbitrage, sa valeur à la date $t$ doit être au moins celle de $S_t$ :
\[
\Pi_t \;\ge\; S_t.
\]
Autrement dit,
\[
C_t + K\,B(t,T) \;\ge\; S_t
\qquad\Longrightarrow\qquad
C_t \;\ge\; S_t - K\,B(t,T).
\]
Comme $B(t,T)=e^{-r(T-t)}\le 1$, on a $K\,B(t,T)\le K$, donc
\[
S_t - K\,B(t,T) \;\ge\; S_t - K.
\]
Finalement,
\[
C_t \;\ge\; S_t - K \;\ge\; (S_t-K)_+,
\]
ce qui prouve bien que le call est toujours supérieur (ou égal) à sa valeur intrinsèque.

%------------------------
\subsection*{Q2. Pas d'exercice anticipé optimal pour un call américain sans dividendes}

Soit $C_t^{A}$ le prix d'un call américain (même $K$, même $T$).  
Par définition d'une option américaine (valeur de continuation + possibilité d'exercice immédiat),
\[
C_t^{A} \;\ge\; (S_t-K)_+.
\]

D'après la question (Q1), on a également pour le call européen :
\[
C_t \;\ge\; (S_t-K)_+.
\]

Lorsque l'action ne verse pas de dividendes ($q=0$), l'unique avantage de l'option américaine
serait d'exercer plus tôt. Or exercer à $t<T$ revient à payer immédiatement $K$ pour recevoir $S_t$,
ce qui fait perdre la valeur temps de l'argent : en reportant le paiement de $K$ à $T$,
on conserve la possibilité de placer $K$ au taux sans risque.

Ainsi, l'exercice anticipé d'un call sans dividendes n'apporte aucun gain : la valeur optimale
est obtenue en conservant l'option jusqu'à l'échéance. On en déduit que
\[
C_t^{A} = C_t
\quad\text{et}\quad
\text{l'exercice optimal a lieu à }T.
\]

\section{Pricing d'un call dans un modèle binomial}

\begin{exerciseBox}[Call européen dans un modèle à deux états]

Un stock vaut 100 aujourd’hui. Demain il vaudra 120 avec proba 1/2 ou 80 avec proba 1/2.

On considère un call européen strike 100.

Quel est son prix aujourd’hui ?
\end{exerciseBox}

\subsection*{Solution :}



On considère un modèle binomial à un pas :

\[
S_0 = 100,\qquad S_T\in\{120,\ 80\}.
\]

On note
\[
u=\frac{120}{100}=1.2,\qquad d=\frac{80}{100}=0.8.
\]

On suppose qu'il existe un actif sans risque tel que, entre $0$ et $T$,
\[
B_0=1,\qquad B_T=1+r \quad (r>-1).
\]


on cherche à calculer le prix $C_0$ du call européen de strike $K=100$:
\[
C_0=\frac{1}{1+r}\Big(q(S_u-K)^+ + (1-q)(S_d-K)^+\Big),
\]
où $S_u=120$ et $S_d=80$.

\subsubsection*{1) Calcule de la probabilité risque-neutre $q$}

Sous une mesure risque-neutre $\mathbb{Q}$, le prix actualisé du sous-jacent est une martingale, donc :
\[
S_0 = \frac{1}{1+r}\,\mathbb{E}^{\mathbb{Q}}[S_T].
\]

Or, dans le modèle binomial :
\[
S_T=
\begin{cases}
S_0u & \text{avec probabilité } q,\\
S_0d & \text{avec probabilité } 1-q.
\end{cases}
\]
Donc :
\[
\mathbb{E}^{\mathbb{Q}}[S_T]=qS_0u+(1-q)S_0d=S_0\big(qu+(1-q)d\big).
\]

En remplaçant dans la condition de martingale :
\[
S_0=\frac{1}{1+r}S_0\big(qu+(1-q)d\big).
\]
Comme $S_0>0$, on simplifie :
\[
1=\frac{1}{1+r}\big(qu+(1-q)d\big)
\quad\Longleftrightarrow\quad
1+r=qu+(1-q)d.
\]

On développe :
\[
1+r=qu+d-qd=d+q(u-d).
\]
Donc :
\[
q(u-d)= (1+r)-d
\quad\Longrightarrow\quad
\boxed{q=\frac{(1+r)-d}{u-d}}.
\]

\paragraph{Condition d'absence d'arbitrage}
On doit avoir $0<q<1$, ce qui équivaut à :
\[
d<1+r<u.
\]
Ici :
\[
0.8 < 1+r < 1.2.
\]

\subsubsection*{2) Prix du call}

Le payoff du call à maturité est :
\[
C_T=(S_T-K)^+.
\]
Ici, avec $K=100$ :
\[
(S_u-K)^+=(120-100)^+=20,\qquad (S_d-K)^+=(80-100)^+=0.
\]

Le pricing risque-neutre donne :
\[
C_0=\frac{1}{1+r}\Big(q\cdot 20 + (1-q)\cdot 0\Big)
=\frac{20q}{1+r}.
\]

En remplaçant $q$ :
\[
\boxed{
C_0=\frac{20}{1+r}\cdot \frac{(1+r)-d}{u-d}
}.
\]

Avec $u=1.2$ , $d=0.8$ et $r=0$ :
\[
\boxed{
C_0=10.
}
\]
